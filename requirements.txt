# SmolCrawl Requirements
# ======================
# Python 3.11+ required

# Core Dependencies (required)
# ----------------------------
pydantic>=2.0.0,<3.0.0        # Data validation - v2 required for compatibility
requests>=2.28.0              # HTTP requests for document processing
httpx[http2]>=0.27.0          # Async HTTP client for web crawling
markdownify>=1.1.0            # HTML to Markdown conversion
readabilipy>=0.3.0            # Content extraction (Mozilla Readability)
beautifulsoup4>=4.12.0        # HTML parsing for URL discovery
lxml>=5.0.0                   # Fast XML/HTML parser (beautifulsoup backend)
loguru>=0.7.3                 # Logging
typer>=0.15.2                 # CLI framework
PyYAML>=6.0                   # YAML config support
diskcache>=5.6.3              # Local caching for crawl results

# Search Index Dependencies (optional - for Tantivy search)
# ---------------------------------------------------------
tantivy>=0.22.2               # Full-text search indexing

# Environment/Utilities
# ---------------------
python-dotenv>=1.1.0          # Environment variable loading

# Development Dependencies (optional)
# -----------------------------------
# pytest>=7.0.0               # Testing
# black>=23.0.0               # Code formatting
# mypy>=1.0.0                 # Type checking

# Notes:
# ------
# 1. SmolCrawl uses a custom lightweight crawler (httpx + BeautifulSoup)
#    instead of heavy frameworks like crawlee. This eliminates dependency
#    conflicts and provides better compatibility with Pydantic v2.
#
# 2. On Windows, readabilipy requires Node.js installed for full functionality
#    The code includes automatic Windows compatibility patches
#
# 3. For document processing only (without search indexing), you can skip
#    tantivy - core extraction works with just the core dependencies.
#
# Minimal install for document processing:
# pip install requests httpx markdownify readabilipy pydantic PyYAML loguru typer beautifulsoup4 diskcache
#
# Full install:
# pip install -r requirements.txt
